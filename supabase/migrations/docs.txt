/*
  # ESSA - Eswatini Schools Association Sports Management System
  
  ## Core Schema

  This migration creates the foundation of the ESSA sports management system:

  ### New Tables

  1. `profiles` - Extended user profiles tied to auth.users
     - role: essa_admin | school_admin | teacher | referee
     - links to school if applicable

  2. `schools` - Registered schools under ESSA
     - school code, name, region, contact details
     - assigned admin user

  3. `sport_activities` - Types of sport at a school (e.g., soccer, netball)
     - links to school, assigned teacher

  4. `players` - Student-players registered at a school
     - authentication fields: student ID, year, parent contact, photo URL
     - verified flag

  5. `player_activity` - Links players to sport activities at a school

  6. `draws` - Tournament draws created by ESSA
     - stage (1, 2, 3), sport, dates, participating schools

  7. `draw_matches` - Individual matches within a draw
     - home school, away school, venue, date

  8. `squads` - Selected team for a specific draw match
     - per school per match

  9. `squad_players` - Individual players in a squad with positions

  10. `announcements` - ESSA announcements visible to schools

  11. `player_updates` - Audit log for player data updates

  ### Security
  - RLS enabled on all tables
  - Role-based access via profiles.role
*/

--profiles table
CREATE TABLE IF NOT EXISTS profiles(
  id uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  full_name text NOT NULL DEFAULT '',
  email text NOT NULL DEFAULT '',
  role text NOT NULL DEFAULT 'school_admin' CHECK(role IN('essa_admin', 'school_admin', 'teacher', 'referee')),
  school_id uuid,
  phone text DEFAULT '',
  must_change_password boolean DEFAULT false,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users can view own profile"
  ON profiles FOR SELECT
  TO authenticated
USING(auth.uid() = id);

CREATE POLICY "Users can update own profile"
  ON profiles FOR UPDATE
  TO authenticated
USING(auth.uid() = id)
  WITH CHECK(auth.uid() = id);

CREATE POLICY "ESSA admins can view all profiles"
  ON profiles FOR SELECT
  TO authenticated
USING(
  EXISTS(SELECT 1 FROM profiles p WHERE p.id = auth.uid() AND p.role = 'essa_admin')
);

CREATE POLICY "ESSA admins can insert profiles"
  ON profiles FOR INSERT
  TO authenticated
  WITH CHECK(
  EXISTS(SELECT 1 FROM profiles p WHERE p.id = auth.uid() AND p.role = 'essa_admin')
);

CREATE POLICY "ESSA admins can update all profiles"

USING(

);

--schools table

--sport_activities table
CREATE TABLE IF NOT EXISTS sport_activities(
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  school_id uuid NOT NULL REFERENCES schools(id) ON DELETE CASCADE,
  name text NOT NULL,
  category text NOT NULL DEFAULT 'team' CHECK(category IN('team', 'individual')),
  teacher_id uuid REFERENCES profiles(id) ON DELETE SET NULL,
  is_active boolean DEFAULT true,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

ALTER TABLE sport_activities ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users can view sport activities for their school"
  ON sport_activities FOR SELECT
  TO authenticated
USING(
  EXISTS(SELECT 1 FROM profiles WHERE id = auth.uid() AND school_id = sport_activities.school_id)
    OR EXISTS(SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'essa_admin')
);

CREATE POLICY "School admins can insert sport activities"
  ON sport_activities FOR INSERT
  TO authenticated
  WITH CHECK(
  EXISTS(SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'school_admin' AND school_id = sport_activities.school_id)
    OR EXISTS(SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'essa_admin')
);

CREATE POLICY "School admins and assigned teachers can update sport activities"
  ON sport_activities FOR UPDATE
  TO authenticated
USING(
  EXISTS(SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'school_admin' AND school_id = sport_activities.school_id)
    OR(teacher_id = auth.uid())
    OR EXISTS(SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'essa_admin')
)
  WITH CHECK(
  EXISTS(SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'school_admin' AND school_id = sport_activities.school_id)
    OR EXISTS(SELECT 1 FROM profiles WHERE id = auth.uid() AND role = 'essa_admin')
);

--players table

  date_of_birth date NOT NULL,

  created_at timestamptz DEFAULT now(),

  UNIQUE(school_id, student_id, enrollment_year)
